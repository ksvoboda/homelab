# Create Debian VM template
qm create 9001 --name "debian-k3s-template" --memory 8192 --cores 2 --net0 virtio,bridge=vmbr0 &&
qm importdisk 9001 debian-13-generic-amd64.qcow2 local-lvm &&
qm set 9001 --scsihw virtio-scsi-pci --scsi0 local-lvm:vm-9001-disk-0 &&
qm set 9001 --ide2 shared:cloudinit &&
qm set 9001 --boot c --bootdisk scsi0 &&
qm set 9001 --serial0 socket --vga serial0 &&
qm set 9001 --agent enabled=1 &&
qm template 9001 

# Once template is cloned, apply appropriate snippet to VM
qm set 103 --cicustom "user=shared:snippets/k3s-init.yaml"
